# TimeMod documents these four country flags as MOD interfaces. Do not set
# forbid_all_crisis_progression, and never clear flags owned by another mod.
pf28_set_crisis_interface_flags = {
 set_country_flag = forbid_become_the_crisis
 set_country_flag = forbid_cosmogenesis
 set_country_flag = forbid_behemoths
 set_country_flag = forbid_galactic_hyperthermia
}



# Keep live crisis interfaces synchronized during ordinary save/load.
pf28_refresh_crisis_interfaces = {
 if = { limit = { pf24_void_pact_bound = yes } pf28_set_crisis_interface_flags = yes }
}
