pf_solaris_flavor_monthly = {
 every_owned_planet = {
  limit = {
   has_planet_flag = pf_solaris
  }
  planet = {
   pf_solaris_flavor_refresh = yes
   if = {
    limit = { is_colony = yes }
    if = {
     limit = { NOT = { has_planet_flag = pf54_flavor_clock } }
     set_planet_flag = pf54_flavor_clock
     set_timed_planet_flag = { flag = pf_flavor_cooldown days = 3600 }
    }
    else_if = {
     limit = { NOT = { has_planet_flag = pf_flavor_cooldown } }
     pf_pick_flavor_event = yes
    }
   }
  }
 }
 if = {
  limit = {
   any_owned_planet = {
    has_planet_flag = pf_solaris
    planet = { is_colony = yes has_modifier = pf_institute_rebuilt }
   }
  }
  if = {
   limit = { NOT = { has_modifier = pf_institute_empire } }
   add_modifier = { modifier = pf_institute_empire days = -1 }
  }
 }
 else = { remove_modifier = pf_institute_empire }
 pf_refresh_blackshore = yes
 if = {
  limit = {
   has_country_flag = pf_ame_secret_ack
   any_owned_planet = {
    has_planet_flag = pf_solaris
    is_controlled_by = owner
   }
  }
  if = {
   limit = {
    NOT = {
     has_modifier = pf_ghost_fleet
    }
   }
   add_modifier = {
    modifier = pf_ghost_fleet
    days = -1
   }
  }
 }
 else = {
  remove_modifier = pf_ghost_fleet
 }
 if = {
  limit = {
   any_owned_planet = {
    has_planet_flag = pf_festival_started
    NOT = {
     has_planet_flag = pf_festival_done
    }
    NOT = {
     has_planet_flag = pf_festival_dialogue_open
    }
   }
  }
  pf_resume_festival = yes
 }
 pf_refresh_array_platforms = yes
}
# Planet scope. The old broad +50% relic-world bonus is replaced, not stacked.
pf_solaris_flavor_refresh = { pf_migrate_restored_features = yes pf17_add_ruins = yes
 if = { limit = { NOT = { has_planet_flag = pf_flavor_initialized } }
  set_planet_flag = pf_flavor_initialized set_variable = { which = pf_flavor_count value = 0 }
  remove_modifier = pm_Phrolova_unstable remove_modifier = pm_Phrolova_beyond remove_modifier = pf_archives_restored
  if = { limit = { NOT = { has_planet_flag = pf_echo_calibrated } NOT = { has_planet_flag = pf_echo_ended } NOT = { has_planet_flag = pf_b_forge_stabilized } }
   add_modifier = { modifier = pf_echo_uncalibrated days = -1 }
  }
 }
 if = { limit = { is_colony = yes NOT = { has_planet_flag = pf_b_forge_stabilized } NOT = { has_modifier = pf_echo_tides } } add_modifier = { modifier = pf_echo_tides days = -1 } }
 # Migrate both pre-merge saves and decoded planets after ownership changes.
 if = { limit = { has_planet_flag = pf075_home_reborn }
  remove_modifier = pf_solaris_memory remove_modifier = pf_old_memories
  if = { limit = { NOT = { has_modifier = pf_solaris_home_reborn } } add_modifier = { modifier = pf_solaris_home_reborn days = -1 } }
 }
 else_if = { limit = { OR = { has_planet_flag = pf_solaris_memory_decoded has_modifier = pf_old_memories owner = { has_country_flag = pf_solaris_ch7 } } }
  set_planet_flag = pf_solaris_memory_decoded
  remove_modifier = pf_solaris_memory
  if = { limit = { NOT = { has_modifier = pf_old_memories } } add_modifier = { modifier = pf_old_memories days = -1 } }
 }
 else = {
  if = { limit = { NOT = { has_modifier = pf_solaris_memory } } add_modifier = { modifier = pf_solaris_memory days = -1 } }
 }
 if = { limit = { pf_three_zones = yes NOT = { has_modifier = pf_institute_rebuilt } }
  add_modifier = { modifier = pf_institute_rebuilt days = -1 }
 }
 if = { limit = { owner = { has_country_flag = pf_ame_secret_ack } NOT = { has_modifier = pf_ghost_frequency } }
  add_modifier = { modifier = pf_ghost_frequency days = -1 }
 }
 if = { limit = { owner = { has_country_flag = pf_aemeath_recruited } NOT = { has_planet_flag = pf_echo_ended } }
  set_planet_flag = pf_echo_ended remove_modifier = pf_echo_uncalibrated remove_modifier = pf_echo_relief
  add_modifier = { modifier = pf_echo_final days = -1 }
 }
}
pf_refresh_blackshore = {
 every_owned_planet = { limit = { has_planet_flag = pf_solaris } planet = {
  if = { limit = { is_controlled_by = owner exists = leader leader = { has_leader_flag = pf_shorekeeper leader_class = official } }
   if = { limit = { NOT = { has_modifier = pf_blackshore_watch } } add_modifier = { modifier = pf_blackshore_watch days = -1 } }
  }
  else = { remove_modifier = pf_blackshore_watch }
 } }
 if = { limit = { any_owned_planet = { has_planet_flag = pf_solaris planet = { is_controlled_by = owner exists = leader leader = { has_leader_flag = pf_shorekeeper leader_class = official } } } }
  if = { limit = { NOT = { has_modifier = pf_blackshore_empire } } add_modifier = { modifier = pf_blackshore_empire days = -1 } }
 }
 else = { remove_modifier = pf_blackshore_empire }
}

 if = { limit = { exists = event_target:pf_solaris event_target:pf_solaris = { has_planet_flag = pf_festival_started NOT = { has_planet_flag = pf_festival_done } } }
  if = { limit = { NOT = { has_country_flag = pf22_festival_intro_seen } NOT = { has_country_flag = pf_festival_talk_1 } } country_event = { id = Phrolova_event.1860 } }
  else_if = { limit = { NOT = { has_country_flag = pf_festival_talk_1 } } country_event = { id = Phrolova_event.1770 } }
