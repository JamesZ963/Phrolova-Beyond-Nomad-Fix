    if = {
        limit = {
            has_country_flag = pf_denia_bond_pending
        }
        remove_country_flag = pf_denia_bond_pending
        if = {
            limit = {
                NOT = {
                    has_country_flag = pf_denia_bond_1
                }
            }
            set_country_flag = pf_denia_bond_1
        }
        else_if = {
            limit = {
                NOT = {
                    has_country_flag = pf_denia_bond_2
                }
            }
            set_country_flag = pf_denia_bond_2
        }
        else_if = {
            limit = {
                NOT = {
                    has_country_flag = pf_denia_bond_3
                }
            }
            set_country_flag = pf_denia_bond_3
            pf17_journal_refresh = yes
        }
        add_resource = {
            energy = 2000
            unity = 500
        }
        set_timed_country_flag = {
            flag = pf_denia_bond_wait
            days = 720
        }
    }
}
