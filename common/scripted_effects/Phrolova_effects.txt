create_Phrolova_leader = {
 if = {
  create_leader = {
   background_ethic = ethic_egalitarian
   class = scientist
   tier = leader_tier_legendary
   species = root
   name = NAME_Phrolova
   custom_description = Phrolova_backstory
   immortal = yes
   hide_age = yes
   gender = female
   skill = 6
   randomize_traits = no
   skip_background_generation = yes
   traits = {
    trait = leader_trait_Phrolova_X
    trait = leader_trait_Phrolova_Z
    trait = leader_trait_Phrolova_Y
   }
   effect = {
    change_leader_portrait = Phrolova0
    set_leader_flag = Phrolova_scientist
    set_leader_flag = leader_death_events_blocked
    set_leader_flag = funeral_option_negative_traits
    set_leader_flag = block_homeworld_traits
   }
  }
  last_created_leader = {
   set_owner = root
   if = {
    limit = {
     owner = {
      has_country_flag = Phrolova_finale_received
     }
    }
    change_leader_portrait = Phrolova1
   }
  }
  limit = {
   NOT = {
    any_owned_leader = {
     pf_guard_id_phrolova = yes
     NOT = {
      is_pool_leader = yes
     }
    }
   }
  }
 }
}
add_Phrolova_leader_ruler = {
    random_owned_leader = {
        limit = { has_leader_flag = Phrolova_scientist }
        set_leader_flag = pf_player_appointed_ruler
        root = { set_leader = prev }
    }
}
remove_Phrolova_leader_ruler = {
    every_owned_leader = { limit = { has_leader_flag = Phrolova_scientist } remove_leader_flag = pf_player_appointed_ruler }
    random_owned_leader = {
        limit = { NOT = { has_leader_flag = Phrolova_scientist } }
        root = { set_leader = prev }
    }
}
pf_find_targets = {
    if = {
        limit = {
            is_scope_type = country
        }

        save_event_target_as = pf_owner

        # After migration, Solaris is the capital Arkship.
        if = {
            limit = {
                has_country_flag = pf_solaris_migrated_to_arkship
                exists = capital_scope
            }

            capital_scope = {
                save_event_target_as = pf_solaris
            }
        }

        # Before migration, locate the original Solaris planet.
        else = {
            random_system = {
                limit = {
                    any_system_planet = {
                        has_planet_flag = pf_solaris@event_target:pf_owner
                    }
                }

                random_system_planet = {
                    limit = {
                        has_planet_flag = pf_solaris@event_target:pf_owner
                    }
                    save_event_target_as = pf_solaris
                }
            }
        }

        # Aleph lookup remains unchanged.
        random_system = {
            limit = {
                has_star_flag = pf_aleph@event_target:pf_owner
            }

            save_event_target_as = pf_aleph_system

            random_system_planet = {
                limit = {
                    has_planet_flag = pf_aleph@event_target:pf_owner
                }
                save_event_target_as = pf_aleph
            }
        }
    }
}

 }
}

pf_check_alignment = {
 pf_find_targets = yes
 if = { limit = { NOT = { has_country_flag = pf_aleph_created } NOT = { has_country_flag = pf_project_open } }
  if = { limit = { exists = event_target:pf_solaris event_target:pf_solaris = { pf_three_zones = yes } }
   if = { limit = { NOT = { has_country_flag = pf_alignment_started } }
    set_country_flag = pf_alignment_started
    set_timed_country_flag = { flag = pf_alignment_wait days = 60 }
    country_event = { id = Phrolova_event.410 days = 60 }
    country_event = { id = Phrolova_event.419 days = 1 }
   }
  }
  else = { remove_country_flag = pf_alignment_started remove_country_flag = pf_alignment_wait remove_country_flag = pf_alignment_ready }
 }
}
