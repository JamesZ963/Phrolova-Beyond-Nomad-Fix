pf_guard_reconcile_phrolova = {
 if = {
  limit = {
   has_country_flag = researched_leaders_Phrolova
  }
  if = {
   limit = {
    NOT = {
     AND = {
      exists = event_target:pf_guard_primary_phrolova@root
      event_target:pf_guard_primary_phrolova@root = {
       pf_guard_id_phrolova = yes
       exists = owner
       owner = {
        is_same_value = root
       }
       NOT = {
        is_pool_leader = yes
       }
      }
     }
    }
   }
   if = {
    limit = {
     NOT = {
      any_owned_leader = {
       pf_guard_id_phrolova = yes
       NOT = {
        is_pool_leader = yes
       }
      }
     }
    }
    if = {
     limit = {
      exists = event_target:pf_guard_primary_phrolova@root
     }
     clone_leader = {
      target = event_target:pf_guard_primary_phrolova@root
      immortal = yes
      skip_background_generation = yes
     }
     last_created_leader = {
      set_owner = root
      remove_leader_flag = pf_guard_retired
      save_global_event_target_as = pf_guard_primary_phrolova@root
     }
    }
    else = {
     create_Phrolova_leader = yes
    }
   }
   random_owned_leader = {
    limit = {
     pf_guard_id_phrolova = yes
     NOT = {
      is_pool_leader = yes
     }
    }
    set_leader_flag = Phrolova_scientist
    save_global_event_target_as = pf_guard_primary_phrolova@root
   }
  }
  if = {
   limit = {
    exists = event_target:pf_guard_primary_phrolova@root
   }
   every_owned_leader = {
    limit = {
     pf_guard_id_phrolova = yes
     NOT = {
      is_same_value = event_target:pf_guard_primary_phrolova@root
     }
    }
    set_leader_flag = pf_guard_retired
    remove_leader_flag = Phrolova_scientist
    kill_leader = {
     show_notification = no
    }
   }
   every_pool_leader = {
    limit = {
     pf_guard_id_phrolova = yes
     NOT = {
      is_same_value = event_target:pf_guard_primary_phrolova@root
     }
    }
    set_leader_flag = pf_guard_retired
    remove_leader_flag = Phrolova_scientist
    kill_leader = {
     show_notification = no
    }
   }
   event_target:pf_guard_primary_phrolova@root = {
    set_leader_flag = Phrolova_scientist
    set_leader_flag = leader_death_events_blocked
   }
  }
  pf_apply_portrait = yes
 }
}
pf_guard_restore_phrolova = {
 if = { limit = { has_country_flag = researched_leaders_Phrolova from = { pf_guard_id_phrolova = yes } }
  if = { limit = { NOT = { any_owned_leader = { pf_guard_id_phrolova = yes NOT = { is_same_value = from } NOT = { is_pool_leader = yes } } } }
   clone_leader = { target = from immortal = yes skip_background_generation = yes }
   last_created_leader = { set_owner = root set_leader_flag = Phrolova_scientist remove_leader_flag = pf_guard_retired
    set_leader_flag = leader_death_events_blocked save_global_event_target_as = pf_guard_primary_phrolova@root }
  }
  from = { set_leader_flag = pf_guard_retired remove_leader_flag = Phrolova_scientist }
 }
}

 }
