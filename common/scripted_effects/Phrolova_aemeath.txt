 if = { limit = { has_country_flag = pf_ame_secret_ack NOT = { has_country_flag = pf_ame_frequency_done } pf_aemeath_location_ready = yes }

pf_begin_ame_contact = {

 pf_find_targets = yes
 if = { limit = { has_country_flag = pf_ame_frequency_done NOT = { has_country_flag = pf_ame_first_contact_done } exists = event_target:pf_solaris }
  random_country = { limit = { is_country_type = pf_aemeath_signal has_country_flag = pf_ame_signal@event_target:pf_owner } save_event_target_as = pf_ame_signal }
  if = { limit = { NOT = { exists = event_target:pf_ame_signal } }
   create_country = { name = NAME_PF_Aemeath_Signal type = pf_aemeath_signal contact_rule = does_first_contact_sites }
   last_created_country = { set_country_flag = pf_ame_signal@event_target:pf_owner save_event_target_as = pf_ame_signal }
  }
  if = { limit = { NOT = { any_active_first_contact = { contact_country = { is_same_value = event_target:pf_ame_signal } } } }
   establish_contact = { who = event_target:pf_ame_signal location = event_target:pf_solaris.solar_system }
  }
 }
}
