 if = { limit = { owner = { has_country_flag = pf_defense_full } } add_modifier = { modifier = pf_local_full days = -1 } }
 else_if = { limit = { owner = { has_country_flag = pf_defense_standard } } add_modifier = { modifier = pf_local_standard days = -1 } }
  else = { add_modifier = { modifier = pf_local_emergency days = -1 } }
 } } }
