  AND = { has_country_flag = pf_defense_standard OR = { NOT = { has_resource = { type = energy amount >= 100 } } NOT = { has_resource = { type = alloys amount >= 50 } } } }
 } }

# Country scope, after pf_find_targets. Time-limited so a lost base cannot retain this support indefinitely.
pf_refresh_solaris_platform_construction = {
 if = {
  limit = { exists = event_target:pf_solaris }
  set_variable = { which = pf_platform_construction_tier value = 0 }
  if = {
   limit = { OR = { pf_solaris_a_defense_active = yes pf_solaris_b_defense_active = yes } }
   set_variable = { which = pf_platform_construction_tier value = 2 }
   if = {
    limit = { OR = {
     AND = { pf_solaris_a_defense_active = yes has_country_flag = pf_defense_full }
     AND = { pf_solaris_b_defense_active = yes check_variable = { which = pf_b19_prepare_days value >= 80 } }
    } }
    set_variable = { which = pf_platform_construction_tier value = 4 }
   }
   else_if = {
    limit = { OR = {
     AND = { pf_solaris_a_defense_active = yes has_country_flag = pf_defense_standard }
     AND = { pf_solaris_b_defense_active = yes check_variable = { which = pf_b19_prepare_days value >= 40 } }
    } }
    set_variable = { which = pf_platform_construction_tier value = 3 }
   }
  }
  event_target:pf_solaris = {
   solar_system = {
    every_starbase_in_system = {
     remove_modifier = pf_defense_platform_construction
     if = {
      limit = {
       is_owned_by = event_target:pf_owner
       is_controlled_by = event_target:pf_owner
       event_target:pf_solaris = { is_owned_by = event_target:pf_owner }
       event_target:pf_owner = { check_variable = { which = pf_platform_construction_tier value > 0 } }
      }
      add_modifier = {
       modifier = pf_defense_platform_construction
       multiplier = owner.pf_platform_construction_tier
       days = 32
      }
     }
    }
   }
  }
 }
