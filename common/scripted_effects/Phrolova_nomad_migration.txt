# ============================================================
# Transfer one planet flag to the destination Arkship.
# Called in the original Solaris planet scope.
# ============================================================

pf_nomad_copy_flag = {
    if = {
        limit = {
            has_planet_flag = $FLAG$
        }

        event_target:pf_nomad_arkship = {
            set_planet_flag = $FLAG$
        }

        remove_planet_flag = $FLAG$
    }
}


# ============================================================
# Transfer one deposit to the destination Arkship.
# ============================================================

pf_nomad_copy_deposit = {
    if = {
        limit = {
            has_deposit = $DEPOSIT$
        }

        event_target:pf_nomad_arkship = {
            if = {
                limit = {
                    NOT = {
                        has_deposit = $DEPOSIT$
                    }
                }
                add_deposit = $DEPOSIT$
            }
        }

        remove_deposit = $DEPOSIT$
    }
}


# ============================================================
# Transfer one modifier.
# DAYS = -1 for permanent modifiers.
# ============================================================

pf_nomad_copy_modifier = {
    if = {
        limit = {
            has_modifier = $MODIFIER$
        }

        event_target:pf_nomad_arkship = {
            if = {
                limit = {
                    NOT = {
                        has_modifier = $MODIFIER$
                    }
                }
                add_modifier = {
                    modifier = $MODIFIER$
                    days = $DAYS$
                }
            }
        }

        remove_modifier = $MODIFIER$
    }
}
