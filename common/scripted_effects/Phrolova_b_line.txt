  if = { limit = { NOT = { any_owned_leader = { has_leader_flag = pf_preview_candidate } } }
   create_leader = { background_ethic = ethic_egalitarian class = scientist tier = leader_tier_legendary species = event_target:pf_preview_host name = NAME_PF_MORIN custom_description = PF_morin_bio

   }
  }

   if = { limit = { NOT = { has_trait = leader_trait_pf_morin } } add_trait = { trait = leader_trait_pf_morin show_message = no } }
   if = { limit = { NOT = { has_trait = leader_trait_pf_morin_2 } } add_trait = { trait = leader_trait_pf_morin_2 show_message = no } }
   if = { limit = { NOT = { has_trait = leader_trait_pf_morin_3 } } add_trait = { trait = leader_trait_pf_morin_3 show_message = no } }
   if = { limit = { NOT = { has_trait = leader_trait_cyborg } } add_trait = { trait = leader_trait_cyborg show_message = no } }
   if = { limit = { NOT = { has_trait = leader_trait_pf_morin_injured } } add_trait = { trait = leader_trait_pf_morin_injured show_message = no } }

  }
 }
}

# B09 is the only research project away from Solaris. There is exactly one
# spatial destination per player: the existing A-line pf_aleph system/asteroid.
# A B-line expedition creates that same site when absent; it never creates a
# second pf_b_aleph site.
pf_b09_find_target = {
    pf_find_targets = yes
    if = {
        limit = {
            exists = event_target:pf_aleph
        }
        event_target:pf_aleph = {
            save_event_target_as = pf_b09_target
        }
    }
    if = {
        limit = {
            exists = event_target:pf_aleph_system
        }
        event_target:pf_aleph_system = {
            random_system_ambient_object = {
                limit = {
                    is_ambient_object_type = pf_tunnel_derelict
                }
                save_event_target_as = pf_b09_wreck
            }
        }
    }
}
