pf46_offer_ancient_pd = {
 if = {
  limit = {
   NOT = {
    has_country_flag = pf46_offered_ancient_pd
   }
  }
  set_country_flag = pf46_offered_ancient_pd
  if = {
   limit = {
    NOT = {
     has_technology = tech_pf_aemeath_ancient_pd
    }
   }
   add_research_option = tech_pf_aemeath_ancient_pd
   add_tech_progress = {
    tech = tech_pf_aemeath_ancient_pd
    progress = 0.25
   }
  }
 }
}
