# Rough tuning knobs; actual fleet power is read in game, not asserted here.

pf_b11_spawn_raid = {
 pf_find_targets = yes
 if = { limit = { any_country = { has_country_flag = pf_b_raid_enemy has_country_flag = pf_b_raid_owner@event_target:pf_owner any_owned_fleet = { has_fleet_flag = pf_b_raid_enemy_fleet } } NOT = { has_country_flag = pf_b11_raid_spawned } }
  set_country_flag = pf_b11_raid_spawned
 set_country_flag = pf_b_raid_wave_1
  set_timed_country_flag = { flag = pf_b11_raid_deadline days = 180 }
  country_event = { id = Phrolova_event.9451 days = 1 }
 }
 if = { limit = {
   NOT = { any_country = { has_country_flag = pf_b_raid_enemy has_country_flag = pf_b_raid_owner@event_target:pf_owner any_owned_fleet = { has_fleet_flag = pf_b_raid_enemy_fleet } } }
   exists = event_target:pf_solaris
   event_target:pf_solaris = { pf_solaris_owner = yes }
   NOT = { has_country_flag = pf_b11_raid_spawned }
   NOT = { has_country_flag = pf_b11_raid_won }
  }
  remove_country_flag = pf_b11_raid_failed
  event_target:pf_solaris = {
   owner = {
    create_country = {
     name = "NAME_PF_B_RAIDERS"
     type = pirate
     species = owner_main_species
     effect = {
      set_country_flag = pf_b_raid_enemy
      set_country_flag = pf_b_raid_owner@event_target:pf_owner
      set_faction_hostility = { set_hostile = yes }
      create_fleet = {
       name = "NAME_PF_B_RAID_FLEET"
       effect = {
        set_owner = prev
        set_location = event_target:pf_solaris
        set_fleet_flag = pf_b_raid_enemy_fleet
        set_fleet_stance = aggressive
        create_ship_design = { design = "NAME_Pirate_Galleon" }
        while = { count = @pf_b_wave1_galleons create_ship = { name = random design = last_created_design graphical_culture = "pirate_01" effect = { set_ship_flag = pf_b_raid_ship } } }
        create_ship_design = { design = "NAME_Black_Earl" }
        while = { count = @pf_b_wave1_cruisers create_ship = { name = random design = last_created_design graphical_culture = "pirate_01" effect = { set_ship_flag = pf_b_raid_ship } } }
        create_ship_design = { design = "NAME_Marauder" }
        while = { count = @pf_b_wave1_destroyers create_ship = { name = random design = last_created_design graphical_culture = "pirate_01" effect = { set_ship_flag = pf_b_raid_ship } } }
       }
      }
     }
    }
   }
  }
  if = { limit = { any_country = { has_country_flag = pf_b_raid_enemy has_country_flag = pf_b_raid_owner@event_target:pf_owner any_owned_fleet = { has_fleet_flag = pf_b_raid_enemy_fleet } } }
   set_country_flag = pf_b11_raid_spawned
 set_country_flag = pf_b_raid_wave_1
   set_timed_country_flag = { flag = pf_b11_raid_deadline days = 180 }
   country_event = { id = Phrolova_event.9451 days = 1 }
  }
 }
}

# ---------- B14: three-stage Star Guild archive site on the mystery system ----------
pf_b14_spawn_site = {
 
 if = { limit = { is_scope_type = country }
 save_event_target_as = pf_owner
 # Local event targets are reacquired from persistent country-linked flags.
 random_system = {
  limit = { any_system_planet = { has_planet_flag = pf_solaris@event_target:pf_owner } }
  random_system_planet = { limit = { has_planet_flag = pf_solaris@event_target:pf_owner } save_event_target_as = pf_solaris }
 }
 random_system = {
  limit = { has_star_flag = pf_aleph@event_target:pf_owner }
  save_event_target_as = pf_aleph_system
  random_system_planet = { limit = { has_planet_flag = pf_aleph@event_target:pf_owner } save_event_target_as = pf_aleph }
 }

 }

 if = { limit = {
   exists = event_target:pf_aleph_system
   event_target:pf_aleph_system = { has_star_flag = pf_aleph@event_target:pf_owner }
   NOT = { event_target:pf_aleph_system = { any_system_planet = { has_planet_flag = pf_b14_site_planet } } }
  }
  event_target:pf_aleph_system = {
   random_system_planet = {
    limit = {
     OR = { is_planet_class = pc_broken is_planet_class = pc_shattered }
     NOT = { has_owner = yes }
     NOT = { is_colony = yes }
     NOT = { has_planet_flag = pf_aleph_asteroid }
     NOT = { has_planet_flag = pf_b14_site_planet }
     NOT = { exists = archaeological_site }
    }
    set_planet_flag = pf_b14_site_planet
    set_planet_flag = pf_b14_site_created
    create_archaeological_site = pf_b_archive_site
   }
  }
 }
 if = { limit = {
   exists = event_target:pf_aleph_system
   event_target:pf_aleph_system = { has_star_flag = pf_aleph@event_target:pf_owner any_system_planet = { has_planet_flag = pf_b14_site_planet NOT = { exists = archaeological_site } } }
  }
  event_target:pf_aleph_system = {
   random_system_planet = {
    limit = { has_planet_flag = pf_b14_site_planet NOT = { exists = archaeological_site } }
    create_archaeological_site = pf_b_archive_site
   }
  }
 }
}
