pf23_schedule_year = {
 set_timed_country_flag = { flag = pf23_specimen_year_wait days = 360 }
 country_event = { id = Phrolova_event.1900 days = 360 }
}
pf23_init_collection = {
 if = {
  limit = {
   is_ai = no
   has_grand_archive_dlc = yes
   OR = {
    pf40_personal_ending_seen = yes
    has_country_flag = pf23_collection_started
   }
  }
  if = {
   limit = {
    NOT = {
     has_country_flag = pf23_collection_started
    }
   }
   set_country_flag = pf23_collection_started
   pf23_schedule_year = yes
  }
  else_if = {
   limit = {
    NOT = {
     has_country_flag = pf23_specimen_year_wait
    }
   }
   set_timed_country_flag = {
    flag = pf23_specimen_year_wait
    days = 1
   }
   country_event = {
    id = Phrolova_event.1900
    days = 1
   }
  }
 }
}
pf23_choose_site = {
 remove_country_flag = pf23_site_found
 random_system = { limit = { is_owned_by = root exists = starbase starbase = { is_controlled_by = root }
  any_system_planet = { is_star = no is_colony = no has_anomaly = no OR = { is_planet_class = pc_barren is_planet_class = pc_barren_cold is_planet_class = pc_asteroid is_planet_class = pc_ice_asteroid is_planet_class = pc_rare_crystal_asteroid is_planet_class = pc_frozen is_planet_class = pc_molten is_planet_class = pc_toxic is_planet_class = pc_gas_giant } NOT = { has_planet_flag = pf_solaris } NOT = { has_planet_flag = pf_aleph_asteroid } }
 }
 random_system_planet = { limit = { is_star = no is_colony = no has_anomaly = no OR = { is_planet_class = pc_barren is_planet_class = pc_barren_cold is_planet_class = pc_asteroid is_planet_class = pc_ice_asteroid is_planet_class = pc_rare_crystal_asteroid is_planet_class = pc_frozen is_planet_class = pc_molten is_planet_class = pc_toxic is_planet_class = pc_gas_giant } NOT = { has_planet_flag = pf_solaris } NOT = { has_planet_flag = pf_aleph_asteroid } }
  save_event_target_as = pf23_site event_target:pf23_owner = { set_country_flag = pf23_site_found }
 }
 }
 if = { limit = { NOT = { has_country_flag = pf23_site_found } exists = capital_scope }
 capital_scope = { solar_system = { spawn_planet = {
  class = pc_asteroid size = 5 name = NAME_PF23_Archive_Asteroid
  orbit_distance = 350 orbit_angle = 210
  init_effect = { save_event_target_as = pf23_site set_planet_flag = pf23_archive_asteroid
   event_target:pf23_owner = { set_country_flag = pf23_site_found }
  }
 } } }
 }
}
