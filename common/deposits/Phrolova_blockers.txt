d_pf_sealed_archive = {
 is_for_colonizable = yes
 category = deposit_cat_blockers
 icon = d_pf_sealed_archive
 time = 120
 resources = {
  category = deposit_blockers
  cost = {
   alloys = 150
  }
 }
 planet_modifier = {
  planet_max_districts_add = -1
 }
 potential = {
  has_planet_flag = pf_solaris
 }
 drop_weight = {
  weight = 0
 }
 can_be_cleared = {
  has_planet_flag = pf_solaris
 }
 on_cleared = {
  if = {
   limit = {
    has_planet_flag = pf_solaris
   }
   if = {
    limit = {
     NOT = {
      has_planet_flag = pf_tower_cleared
     }
    }
    set_planet_flag = pf_tower_cleared
    if = {
     limit = {
      owner = {
       is_gestalt = yes
      }
     }
     if = {
      limit = {
       NOT = {
        has_deposit = d_pf_tower_restored
       }
      }
      add_deposit = d_pf_tower_restored
     }
    }
    else = {
     if = {
      limit = {
       NOT = {
        has_deposit = d_pf_tower_restored
       }
      }
      add_deposit = d_pf_tower_restored
     }
    }
   }
  }
 }
}
