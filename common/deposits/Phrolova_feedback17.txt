d_pf_solaris_ruin_01 = {
 is_for_colonizable = yes
 category = deposit_cat_blockers
 time = 60
 resources = {
  category = deposit_blockers
  cost = {
   energy = 100
   minerals = 100
  }
 }
 planet_modifier = {
  planet_amenities_add = -1
 }
 potential = {
  has_planet_flag = pf_solaris
 }
 drop_weight = {
  weight = 0
 }
 can_be_cleared = {
  has_planet_flag = pf_solaris
 }
 on_cleared = {
  if = {
   limit = {
    has_planet_flag = pf_solaris
   }
   if = {
    limit = {
     NOT = {
      has_planet_flag = pf_ruin_01_restored
     }
    }
    set_planet_flag = pf_ruin_01_restored
    if = {
     limit = {
      owner = {
       is_gestalt = yes
      }
     }
     if = {
      limit = {
       NOT = {
        has_deposit = d_pf_ruin_01_restored
       }
      }
      add_deposit = d_pf_ruin_01_restored
     }
    }
    else = {
     if = {
      limit = {
       NOT = {
        has_deposit = d_pf_ruin_01_restored
       }
      }
      add_deposit = d_pf_ruin_01_restored
     }
    }
   }
  }
 }
 icon = d_central_spire
}
