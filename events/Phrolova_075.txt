country_event = { id = Phrolova_event.11060 is_triggered_only = yes hide_window = yes trigger = { is_ai = no } immediate = { 
 save_event_target_as = pf_owner
 if = { limit = { exists = fromfrom } fromfrom = { remove_fleet_flag = pf075_aid_active } }
 
if = { limit = { from = { pf075_crisis_country = yes } } set_country_flag = pf075_crisis_participated }
if = { limit = { from = { is_country_type = swarm } } set_country_flag = pf075_fought_swarm }
if = { limit = { from = { OR = { is_country_type = extradimensional is_country_type = extradimensional_2 is_country_type = extradimensional_3 } } } set_country_flag = pf075_fought_unbidden }
if = { limit = { from = { is_country_type = ai_empire } } set_country_flag = pf075_fought_contingency }
if = { limit = { from = { OR = { is_country_type = synth_queen is_country_type = awakened_synth_queen } } } set_country_flag = pf075_fought_queen }

 if = { limit = { has_country_flag = pf_aemeath_recruited exists = fromfrom exists = fromfromfrom
  fromfrom = { exists = leader leader = { has_leader_flag = pf_aemeath }
   any_owned_ship = { has_ship_flag = pf075_suizhe_origin@root }
   exists = space_owner space_owner = { is_country_type = default NOT = { is_same_value = root } NOT = { is_hostile = root } NOT = { is_at_war_with = root } }
  }
  fromfromfrom = { fleet_power >= 5000 }
 }
  fromfrom = { save_event_target_as = pf075_battle_fleet space_owner = { save_event_target_as = pf075_recipient } }
  if = { limit = { from = { OR = { is_at_war_with = event_target:pf075_recipient is_hostile = event_target:pf075_recipient } } }
   every_country = { event_target:pf075_battle_fleet = { remove_fleet_flag = pf075_aid_for@prev } }
   event_target:pf075_battle_fleet = { set_fleet_flag = pf075_aid_for@event_target:pf075_recipient set_fleet_flag = pf075_aid_active }
  }
 }
 } }

country_event = {
 id = Phrolova_event.11061
 is_triggered_only = yes
 hide_window = yes
 trigger = {
  is_ai = no
 }
 immediate = {
  save_event_target_as = pf_owner
  if = {
   limit = {
    exists = fromfrom
    fromfrom = {
     has_fleet_flag = pf075_aid_active
     exists = leader
     leader = {
      has_leader_flag = pf_aemeath
     }
     any_owned_ship = {
      has_ship_flag = pf075_suizhe_origin@root
     }
    }
   }
   fromfrom = {
    save_event_target_as = pf075_battle_fleet
    remove_fleet_flag = pf075_aid_active
   }
   random_country = {
    limit = {
     event_target:pf075_battle_fleet = {
      has_fleet_flag = pf075_aid_for@prev
     }
    }
    save_event_target_as = pf075_recipient
   }
   if = {
    limit = {
     exists = event_target:pf075_recipient
     event_target:pf075_recipient = {
      NOT = {
       has_country_flag = pf075_aid_cd@root
      }
     }
    }
    event_target:pf075_recipient = {
     set_timed_country_flag = {
      flag = pf075_aid_cd@event_target:pf_owner
      days = 720
     }
     set_timed_country_flag = {
      flag = pf075_aid_goodwill@event_target:pf_owner
      days = 7200
     }
    }
    if = {
     limit = {
      NOT = {
       has_country_flag = pf075_aid_trust_received
      }
     }
     set_country_flag = pf075_aid_trust_received
     add_modifier = {
      modifier = pf075_aid_trust
      days = -1
     }
    }
    if = {
     limit = {
      event_target:pf075_recipient = {
       NOT = {
        has_country_flag = pf075_aid_seen@root
       }
      }
     }
     event_target:pf075_recipient = {
      set_country_flag = pf075_aid_seen@event_target:pf_owner
     }
     change_variable = {
      which = pf075_distinct_aid
      value = 1
     }
    }
    pf075_pay_integrity = yes
    pf075_sync_aid_trait = yes
    if = {
     limit = {
      NOT = {
       has_country_flag = pf075_first_aid_seen
      }
     }
     set_country_flag = pf075_first_aid_seen
     country_event = {
      id = Phrolova_event.11030
     }
    }
   }
  }
 }
}
